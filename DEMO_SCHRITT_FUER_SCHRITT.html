Die zwei T-Beispiele direkt anzeigen

Den gesamten Block im aktivierten Terminal im Repository-Root einfügen. Die CSV wird in Blöcken gelesen und beendet, sobald beide Beispielschlüssel gefunden sind.

python - <<'PY'
from pathlib import Path
import pandas as pd

ROOT = next(p for p in [Path.cwd(), *Path.cwd().parents]
            if (p / "data-manifest.json").is_file())

keys = [("A5000005424", "S101949793", 2020),
        ("A5000000364", "S101949793", 2020)]
cols = ["author_id", "journal_id", "t", "coauthor_seed", "entering_work_id",
        "first_entry_ride", "topic_match", "n_profile_papers", "tm_status"]
parts = []
for chunk in pd.read_csv(ROOT / "C_topic_match/data/event_table_topicmatch_v5.csv",
                         usecols=cols, chunksize=100_000):
    index = pd.MultiIndex.from_frame(chunk[["author_id", "journal_id", "t"]])
    selected = chunk.loc[index.isin(keys)]
    if len(selected):
        parts.append(selected)
    if sum(len(part) for part in parts) == len(keys):
        break
assert parts, "No example rows found"
rows = pd.concat(parts).set_index(["author_id", "journal_id", "t"])
assert len(rows) == len(keys) and rows.index.is_unique, "Example keys differ"
print(rows.loc[keys].to_string())
PY
